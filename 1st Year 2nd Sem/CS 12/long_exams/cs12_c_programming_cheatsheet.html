Pointer Arithmetic
int nums[] = {10, 20, 30};
// These three loops are IDENTICAL:
for (int i=0; i<3; i++) nums[i]++;
for (int i=0; i<3; i++) *(nums+i)++;
for (int*p=nums; p<nums+3; p++) (*p)++;
Array name decays to pointer to first element. p[i] = *(p+i) always.

Return array — WRONG vs RIGHT
// WRONG: returning ptr to local (dangling!)
int* wrong(int n) { int ret[n]; return ret; }

// RIGHT: output parameter (caller owns array)
void range(int *out, int n) {
    for (int i=0; i<n; i++) out[i] = i;
}
int nums[10];
range(nums, 10); // range modifies in place

Dynamic Matrices
Approach 1 — Contiguous (preferred)
// One big malloc: rows × cols elements
void *make_matrix(int r, int c, int sz) {
    return calloc(r * c, sz);
}
// Usage — type: int (*mat)[COLS]
int (*mat)[5] = make_matrix(3, 5, sizeof(int));
mat[1][3] = 42;  // works like 2D array
free(mat);       // single free!
Approach 2 — Non-contiguous (ptr of ptrs)
void **make_nc(int rows, int cols, int sz) {
    void **p = calloc(rows, sizeof(void*));
    for (int i=0; i<rows; i++)
        p[i] = calloc(cols, sz);
    return p;
}
int **mat = (int**)make_nc(3,5,sizeof(int));
mat[1][3] = 42;
// Must free each row + then outer array!
for (int i=0; i<3; i++) free(mat[i]);
free(mat);

Linked Lists
typedef struct node Node;
struct node { int data; Node *next; };
typedef struct { Node *head; int size; } List;

Node *make_node(int data, Node *next) {
    Node *n = malloc(sizeof(Node));
    n->data = data;  n->next = next;
    return n;
}
Build (stack-style shorthand)
// h1 → h2 → h3 → NULL
Node *h3 = make_node(30, NULL);
Node *h2 = make_node(20, h3);
Node *h1 = make_node(10, h2);
Traversal
for (Node *t=head; t!=NULL; t=t->next)
    printf("%d\n", t->data);
Prepend (O(1))
void prepend(List *l, int d) {
    l->head = make_node(d, l->head);
    l->size++;
}
Append (O(n))
void append(List *l, int d) {
    if (!l->head) { l->head=make_node(d,NULL); }
    else {
        Node *t = l->head;
        while (t->next) t = t->next;
        t->next = make_node(d, NULL);
    }
    l->size++;
}
Free entire list
Node *cur = head;
while (cur) { Node *nx=cur->next; free(cur); cur=nx; }
Multithreaded Programming (pthreads)
Thread Lifecycle
// Thread function signature (always void*→void*)
void *fn(void *arg) {
    int *p = arg;   // cast arg to real type
    // ... work ...
    return NULL;
}
pthread_t t1;
pthread_create(&t1, NULL, fn, &n); // spawn
pthread_join(t1, NULL);            // wait/sync
Without pthread_join, main may exit before thread finishes → nondeterministic output.
Passing data to thread (struct pattern)
struct thread_data { int n; int *nums; ... };

struct thread_data d = {.n=0, .nums=nums};
pthread_create(&t1, NULL, f, &d);
Key facts
Global varsShared between all threads
Local varsPrivate to each thread (own stack)
Execution orderNondeterministic on multicore
SpeedupOnly if work is CPU-bound & parallelizable
Race Condition vs Mutex Fix
// RACE: n++ is NOT atomic (3 asm steps!)
int n = 0;
// Thread 1 and 2 both read, increment, write
// → result unpredictable

// FIX: mutex (lock)
pthread_mutex_t lock;
pthread_mutex_init(&lock, NULL);

void *plus(void *arg) {
    for (int i=0; i<10000; i++) {
        pthread_mutex_lock(&lock);
        n++;  // critical section
        pthread_mutex_unlock(&lock);
    }
    return NULL;
}
Performance tip — minimize locking
// BAD: lock/unlock 1B times (huge overhead)
for (i=start; i<end; i++) {
    pthread_mutex_lock(&lk); positive++; pthread_mutex_unlock(&lk);
}

// GOOD: accumulate locally, lock once at end
int local = 0;
for (i=start; i<end; i++) if (nums[i]>0) local++;
pthread_mutex_lock(&lk);
positive += local;
pthread_mutex_unlock(&lk);
K-thread work split pattern
int chunk = N / K;
int start = chunk * thread_id;
int stop  = start + chunk;
for (int i=start; i<stop; i++) { ... }
sync.c bug: locks inside loop per element = slow. k_threads.c fixes it with local accumulation.
two_threads.c bug: positive_local counts negatives and vice versa (condition is flipped).
